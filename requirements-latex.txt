# LaTeX packages required by Inkwell templates.
# These are tlmgr package names (not .sty filenames).
#
# Install all at once:
#   sed 's/#.*//' requirements-latex.txt | xargs tlmgr install && texhash

# core (usually present but TinyTeX minimal installs may lack them)
geometry
hyperref
babel
# English language file. On a full TeX Live / MacTeX install this
# ships with the base `babel` package. On a minimal TinyTeX install it
# ships as a separate `babel-english` package, so rho / rmxaa / any
# template that loads `\usepackage[english]{babel}` fails with
# "Unknown option 'english'" until this is installed.
babel-english
# Spanish language files. Required by the rho and rmxaa templates even
# when the document is authored in English: rhobabel.sty contains
# \iflanguage{spanish}{...}{...} branches, and \iflanguage is a hard
# error when the language is not declared to babel. babel-spanish
# provides the language definition; hyphen-spanish prevents "No
# hyphenation patterns were preloaded for Spanish" warnings from
# turning into run-length pagination drift.
babel-spanish
hyphen-spanish
iftex

# EPS support. The kth-letter template embeds an .eps logo; under
# pdflatex this requires epstopdf.sty (epstopdf-pkg) plus the epstopdf
# conversion script. The script also needs Ghostscript on PATH (ships
# with MacTeX; `brew install ghostscript` / `apt install ghostscript`
# otherwise).
epstopdf-pkg
epstopdf

# hipster-cv template dependencies. The simplehipstercv class draws its
# sidebar, timeline, and info bubbles with TikZ (pgf) and FontAwesome
# icons, sets the body in Raleway, and lays out the two columns with
# paracol. smartdiagram and tikz-3dplot are loaded by the class's style
# file for its bubble-diagram helpers.
fontawesome
raleway
# raleway's .sty defaults to the LY1 font encoding; ly1 provides
# ly1enc.def, which full TeX Live ships but minimal TinyTeX does not
# ("Encoding file 'ly1enc.def' not found" at \ProcessOptions).
ly1
paracol
smartdiagram
tikz-3dplot
pgf
